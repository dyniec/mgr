\begin{frame}
\frametitle{Scheme\cite{scheme} i call/cc}
Operator \texttt{call/cc} łapie całą resztę programu i przekazuje jako parametr.

Gdy ta kontynuacja jest zawołana reszta programu zostanie podmieniona przez nią.
\begin{examples}
\texttt{10 + (call/cc ($\lambda$ k . k (2 + (k 1))))}
\end{examples}
\end{frame}
\begin{frame}
\frametitle{Kontynuacje ograniczone}
\begin{itemize}
\item widać zakres
\item normalne funkcje
\item $\#$ i  $\mathcal {F}$ \cite{prompts}
\item \texttt{shift} i \texttt{reset} \cite{abstracting}
\end{itemize}
\begin{block}{Redukcja}
\texttt{(reset E[(shift k expr)]) \\ => (reset (($\lambda$ k.expr) ($\lambda$ v.(reset E[v]))))}
\end{block}
\begin{examples}
\texttt{2* (reset (+ 1 (shift k (k (k 1)))))}
\end{examples}
\end{frame}
\begin{frame}
\frametitle{$shift_0$ i $reset_0$ \cite{functional_abstraction}}
\begin{block}{Redukcja}
\texttt{(reset$_0$ E[(shift$_0$ k expr)]) \\ => ($\lambda$ k.expr) ($\lambda$ v.(reset$_0$ E[v]))}
\end{block}

\begin{examples}
 \texttt{reset$_0$(3 + reset$_0$(4 * shift$_0$ k in shift$_0$ c in E))} \\
 $k$ $\approx$ \texttt{$\lambda$ x.4*x}\\
 $c$ $\approx$ \texttt{$\lambda$ x.3+x}
\end{examples}
\end{frame}
